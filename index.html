function turboMathTick() { if (!turboActive) return; const now = Date.now(); // 🚀 FREEZE CHECK runs FIRST if (turboLastFrameTime > 0 && turboRateFramesPerSec > 0) { const gap = now - turboLastFrameTime; if (gap > 500) { const missed = Math.floor((gap / 1000) * turboRateFramesPerSec) - 1; if (missed > 0) { const actual = Math.min(missed, CATCHUP_CAP); applyTurboFrames(actual); turboFrames += actual; setTurboInfo("⚡ Caught up " + actual.toLocaleString("en-IN") + " missed Turbo frames.", false, true); } } } // ... do the increment ... turboFrames++; // Update timestamp AFTER the freeze check turboLastFrameTime = now; // ... }
